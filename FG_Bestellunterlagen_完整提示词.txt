import pdfplumber
import pandas as pd
from pathlib import Path
import re

INPUT_FILE = "FG Prüfung Am Ettersberg.pdf"
pdf_path = Path(INPUT_FILE)
OUTPUT_FILE = pdf_path.with_name(pdf_path.stem + "_DaimLER_FGPruefung.xlsx")
CODE_RE = re.compile(r'^[A-Z][A-Z0-9]{2,4}$')
HEADER_WORDS = [
    "Vor Übergabe an Kunden", "Seite",
    "Albert Ziegler GmbH", "Bestellung von"
]
END_WORDS = ["Gesamtpreis Fahrzeug netto", "Kaufpreis"]


def clean(txt):
    txt = txt.replace("\xa0", " ")
    txt = re.sub(r"\s+", " ", txt)
    return txt.strip()


rows = []
with pdfplumber.open(INPUT_FILE) as pdf:
    # 基线配置页范围，不代表增强版全文检查范围
    for page in pdf.pages[1:20]:
        text = page.extract_text()
        if not text:
            continue
        lines = text.split("\n")
        for line in lines:
            line = clean(line)
            if not line:
                continue
            if any(h in line for h in HEADER_WORDS):
                continue
            if any(e in line for e in END_WORDS):
                break
            if line.startswith("**"):
                continue
            m = re.match(r'^([A-Z][A-Z0-9]{2,4})\s+(.+)$', line)
            if not m:
                continue
            code = m.group(1)
            desc = m.group(2)
            if CODE_RE.match(code):
                rows.append([code, desc])

# 保留全部基线记录，不执行去重
df = pd.DataFrame(rows, columns=["Code", "Description"])

df.to_excel(OUTPUT_FILE, index=False, engine="openpyxl")
print(df.head())
print()
print("共识别:", len(df))
print("Excel已生成:", OUTPUT_FILE)

import pandas as pd
from pathlib import Path
from datetime import datetime
import re
import sys

BESTELL_KEYWORDS = ["Daimler_Bestellunterlagen"]
FG_KEYWORDS = [
    "Daimler_FGPruefung", "Daimler_FGPrüfung", "Daimler_FGPrufung"
]


def clean_value(value):
    if pd.isna(value):
        return ""
    value = str(value).replace("\xa0", " ")
    value = re.sub(r"\s+", " ", value)
    return value.strip()


def find_latest_xlsx_by_keywords(keywords):
    # 必须在隔离基线目录运行，避免选到旧文件
    candidates = []
    for p in Path(".").glob("*.xlsx"):
        if p.name.startswith("~$"):
            continue
        name_lower = p.name.lower()
        for kw in keywords:
            if kw.lower() in name_lower:
                candidates.append(p)
                break
    if not candidates:
        raise FileNotFoundError(
            f"未找到名称包含以下关键词的xlsx文件: {keywords}"
        )
    candidates = sorted(
        candidates, key=lambda x: x.stat().st_mtime, reverse=True
    )
    print("找到候选文件:")
    for p in candidates:
        print(" -", p.name)
    print("使用文件:", candidates[0].name)
    print()
    return candidates[0]


def find_column(df, possible_names):
    for name in possible_names:
        if name in df.columns:
            return name
    raise ValueError(
        f"未找到列名。候选列名: {possible_names}，实际列名: {list(df.columns)}"
    )


def build_code_dict(df, code_col, desc_col):
    # 基线仅保留第一次出现的代码及描述
    result = {}
    for _, row in df.iterrows():
        code = clean_value(row.get(code_col, ""))
        if not code:
            continue
        desc = clean_value(row.get(desc_col, ""))
        if code not in result:
            result[code] = desc
    return result


def main():
    bestell_file = find_latest_xlsx_by_keywords(BESTELL_KEYWORDS)
    fg_file = find_latest_xlsx_by_keywords(FG_KEYWORDS)
    print("Bestellunterlagen Excel:", bestell_file.name)
    print("FG Prüfung Excel:", fg_file.name)
    print()

    df_bestell = pd.read_excel(bestell_file, dtype=str, engine="openpyxl")
    df_fg = pd.read_excel(fg_file, dtype=str, engine="openpyxl")

    bestell_code_col = find_column(df_bestell, ["Code", "VCode"])
    bestell_desc_col = find_column(
        df_bestell, ["Description", "Beschreibung", "Name"]
    )
    fg_code_col = find_column(df_fg, ["Code", "VCode"])
    fg_desc_col = find_column(df_fg, ["Description", "Beschreibung", "Name"])

    print("Bestell Code列:", bestell_code_col)
    print("Bestell Description列:", bestell_desc_col)
    print("FG Code列:", fg_code_col)
    print("FG Description列:", fg_desc_col)
    print()

    bestell_dict = build_code_dict(df_bestell, bestell_code_col, bestell_desc_col)
    fg_dict = build_code_dict(df_fg, fg_code_col, fg_desc_col)
    bestell_codes = set(bestell_dict.keys())
    fg_codes = set(fg_dict.keys())
    only_in_bestell = bestell_codes - fg_codes
    only_in_fg = fg_codes - bestell_codes
    matched = bestell_codes & fg_codes

    only_bestell_rows = []
    for code in bestell_dict.keys():
        if code in only_in_bestell:
            only_bestell_rows.append({
                "Code": code, "Description": bestell_dict.get(code, "")
            })
    df_only_bestell = pd.DataFrame(
        only_bestell_rows, columns=["Code", "Description"]
    )

    only_fg_rows = []
    for code in fg_dict.keys():
        if code in only_in_fg:
            only_fg_rows.append({
                "Code": code, "Description": fg_dict.get(code, "")
            })
    df_only_fg = pd.DataFrame(only_fg_rows, columns=["Code", "Description"])

    matched_rows = []
    for code in bestell_dict.keys():
        if code in matched:
            matched_rows.append({
                "Code": code,
                "Bestellunterlagen_Description": bestell_dict.get(code, ""),
                "FGPruefung_Description": fg_dict.get(code, "")
            })
    df_matched = pd.DataFrame(matched_rows, columns=[
        "Code", "Bestellunterlagen_Description", "FGPruefung_Description"
    ])

    detail_rows = []
    all_codes_ordered = []
    for code in fg_dict.keys():
        if code not in all_codes_ordered:
            all_codes_ordered.append(code)
    for code in bestell_dict.keys():
        if code not in all_codes_ordered:
            all_codes_ordered.append(code)

    for code in all_codes_ordered:
        in_fg = code in fg_codes
        in_bestell = code in bestell_codes
        if in_fg and in_bestell:
            status = "MATCHED"
        elif in_fg and not in_bestell:
            status = "ONLY_IN_FG"
        elif in_bestell and not in_fg:
            status = "ONLY_IN_BESTELLUNTERLAGEN"
        else:
            status = "UNKNOWN"
        detail_rows.append({
            "Code": code,
            "FGPruefung_Description": fg_dict.get(code, ""),
            "Bestellunterlagen_Description": bestell_dict.get(code, ""),
            "In_FGPruefung": "Ja" if in_fg else "Nein",
            "In_Bestellunterlagen": "Ja" if in_bestell else "Nein",
            "Status": status
        })
    df_details = pd.DataFrame(detail_rows, columns=[
        "Code", "FGPruefung_Description", "Bestellunterlagen_Description",
        "In_FGPruefung", "In_Bestellunterlagen", "Status"
    ])

    df_summary = pd.DataFrame([
        ["Bestellunterlagen文件", bestell_file.name],
        ["FG Prüfung文件", fg_file.name],
        ["Bestellunterlagen Code数量", len(bestell_codes)],
        ["FG Prüfung Code数量", len(fg_codes)],
        ["Matched数量", len(matched)],
        ["Only in Bestellunterlagen数量", len(only_in_bestell)],
        ["Only in FG Prüfung数量", len(only_in_fg)],
        ["生成时间", datetime.now().strftime("%Y-%m-%d %H:%M:%S")],
    ], columns=["项目", "值"])

    output_file = Path(
        f"Vergleich_{fg_file.stem}_vs_{bestell_file.stem}.xlsx"
    )
    with pd.ExcelWriter(output_file, engine="openpyxl") as writer:
        df_summary.to_excel(writer, sheet_name="Summary", index=False)
        df_only_fg.to_excel(writer, sheet_name="Only_in_FG", index=False)
        df_only_bestell.to_excel(writer, sheet_name="Only_in_Bestell", index=False)
        df_matched.to_excel(writer, sheet_name="Matched", index=False)
        df_details.to_excel(writer, sheet_name="Details", index=False)

    print("比较完成")
    print("Excel已生成:", output_file)
    print()
    print("Bestellunterlagen Code数量:", len(bestell_codes))
    print("FG Prüfung Code数量:", len(fg_codes))
    print("Matched:", len(matched))
    print("Only in Bestellunterlagen:", len(only_in_bestell))
    print("Only in FG Prüfung:", len(only_in_fg))


if __name__ == "__main__":
    try:
        main()
    except Exception as e:
        print("程序出错:")
        print(e)
        sys.exit(1)
